/* ECG over BLE - CJMCU-8232 (AD8232) + ESP32 ------------------------------------------ Membaca sinyal ECG dari AD8232 lewat ADC, lalu mengirimkannya melalui BLE Notify ke web app (Web Bluetooth API). Koneksi hardware: AD8232 OUTPUT -> ESP32 GPIO34 (ADC1, input only) AD8232 LO+ -> ESP32 GPIO32 AD8232 LO- -> ESP32 GPIO33 AD8232 3.3V -> ESP32 3V3 AD8232 GND -> ESP32 GND Format paket BLE (setiap notifikasi): byte 0 : status lead-off (1 = elektroda lepas, 0 = normal) byte 1 : jumlah sampel dalam paket ini (N) byte 2.. : N sampel ADC, masing-masing uint16 little-endian (0-4095) Sampel dikumpulkan dulu dalam buffer lalu dikirim sekaligus per batch supaya jumlah notifikasi BLE per detik tidak terlalu tinggi (BLE punya batas practical notification rate). Sebelum dikirim, tiap sampel melewati filter software (moving average + high-pass) untuk meredam noise ADC dan baseline wander. Penguatan (gain) tampilan diatur di sisi web app agar bisa disesuaikan tanpa perlu flash ulang firmware. */ #include #include #include #include #include // ---------- Konfigurasi pin ---------- const int PIN_ECG = 34; const int PIN_LO_PLUS = 32; const int PIN_LO_MINUS = 33; // ---------- Konfigurasi sampling ---------- const int SAMPLE_INTERVAL_MS = 5; // 200 Hz const int BATCH_SIZE = 10; // kirim tiap 10 sampel (~50ms / batch, 20Hz notify) // ---------- UUID BLE (custom, boleh diganti asal konsisten dgn web app) ---------- #define SERVICE_UUID "6e400001-b5a3-f393-e0a9-e50e24dcca9e" #define CHARACTERISTIC_UUID "6e400003-b5a3-f393-e0a9-e50e24dcca9e" BLEServer *pServer = nullptr; BLECharacteristic *pCharacteristic = nullptr; bool deviceConnected = false; uint16_t sampleBuffer[BATCH_SIZE]; uint8_t sampleCount = 0; uint8_t txPacket[2 + BATCH_SIZE * 2]; // ---------- Filter software (mengurangi noise sebelum dikirim) ---------- // Moving average: meredam noise frekuensi tinggi dari ADC. const int MA_N = 4; uint16_t maBuffer[MA_N] = {2048, 2048, 2048, 2048}; uint8_t maIndex = 0; // High-pass sederhana: menghilangkan baseline wander (sinyal melayang naik-turun // pelan akibat napas/gerakan) sehingga sinyal tetap stabil di sekitar titik tengah. const float HP_ALPHA = 0.98f; // makin dekat ke 1, makin lambat baseline "dilupakan" float hpPrevRaw = 2048.0f; float hpPrevOut = 0.0f; uint16_t filterSample(uint16_t raw) { maBuffer[maIndex] = raw; maIndex = (maIndex + 1) % MA_N; uint32_t sum = 0; for (int i = 0; i < MA_N; i++) sum += maBuffer[i]; float avg = (float)sum / MA_N; float hpOut = HP_ALPHA * (hpPrevOut + avg - hpPrevRaw); hpPrevRaw = avg; hpPrevOut = hpOut; float centered = 2048.0f + hpOut; if (centered < 0) centered = 0; if (centered > 4095) centered = 4095; return (uint16_t)centered; } class ServerCallbacks : public BLEServerCallbacks { void onConnect(BLEServer *server) override { deviceConnected = true; Serial.println("Web app terhubung."); } void onDisconnect(BLEServer *server) override { deviceConnected = false; Serial.println("Web app terputus. Mulai advertising lagi..."); delay(200); server->getAdvertising()->start(); } }; void setupBLE() { BLEDevice::init("ESP32-ECG"); pServer = BLEDevice::createServer(); pServer->setCallbacks(new ServerCallbacks()); BLEService *pService = pServer->createService(SERVICE_UUID); pCharacteristic = pService->createCharacteristic( CHARACTERISTIC_UUID, BLECharacteristic::PROPERTY_NOTIFY); pCharacteristic->addDescriptor(new BLE2902()); pService->start(); BLEAdvertising *pAdvertising = BLEDevice::getAdvertising(); pAdvertising->addServiceUUID(SERVICE_UUID); pAdvertising->setScanResponse(true); BLEDevice::startAdvertising(); Serial.println("BLE siap, menunggu koneksi dengan nama \"ESP32-ECG\"..."); } void setup() { Serial.begin(115200); pinMode(PIN_LO_PLUS, INPUT); pinMode(PIN_LO_MINUS, INPUT); analogReadResolution(12); // 0 - 4095 analogSetAttenuation(ADC_11db); // jangkauan ~0 - 3.3V setupBLE(); } void loop() { static uint32_t lastSample = 0; uint32_t now = millis(); if (now - lastSample >= (uint32_t)SAMPLE_INTERVAL_MS) { lastSample = now; bool leadOff = (digitalRead(PIN_LO_PLUS) == HIGH) || (digitalRead(PIN_LO_MINUS) == HIGH); uint16_t value = leadOff ? 2048 : filterSample(analogRead(PIN_ECG)); sampleBuffer[sampleCount++] = value; if (sampleCount >= BATCH_SIZE) { if (deviceConnected) { txPacket[0] = leadOff ? 1 : 0; txPacket[1] = sampleCount; for (int i = 0; i < sampleCount; i++) { txPacket[2 + i * 2] = sampleBuffer[i] & 0xFF; txPacket[2 + i * 2 + 1] = (sampleBuffer[i] >> 8) & 0xFF; } pCharacteristic->setValue(txPacket, 2 + sampleCount * 2); pCharacteristic->notify(); } sampleCount = 0; } } }
